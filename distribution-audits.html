function calculateAuditScore(){ let scores=document.querySelectorAll(".score"); let total=0; let max=0; scores.forEach(select=>{ let value=parseInt(select.value); if(value>0){ total+=value; max+=3; } }); let percent=Math.round((total/max)*100); document.getElementById("auditScore").innerText=percent+"%"; generateRating(percent); } function generateRating(percent){ let rating=""; if(percent>=90){ rating="Excellent Compliance"; } else if(percent>=75){ rating="Good Compliance"; } else if(percent>=60){ rating="Moderate Risk – Improvements Required"; } else{ rating="High Risk – Immediate Action Required"; } document.getElementById("auditRating").innerText=rating; }
